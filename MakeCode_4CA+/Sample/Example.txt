１．コメントアウト

MUGENと同じく、";"によってコメントアウトができます。

 例

 MOV DWORD [ESI+EDX], ECX;	Aliveに0を代入

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
２．マクロ定数定義 ※濫用注意※

%define <定数名> <値>

という構文により、定数を定義することができます。

 例

 %define ESCkey 4942209
 ︙
 MOV DWORD [ESCkey], ECX;	ESCキー押下フラグを0にする

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
３．マクロ関数定義 ※濫用注意※

%macro <関数名> <引数の数>
	<関数処理>
%endmacro

という構文により、マクロ関数を定義することができます。

 例

 %macro strlen 1	;引数として渡した文字列の長さを図るマクロ関数
	PUSH EDI
 	MOV EDI, %1
	XOR EAX, EAX
	XOR ECX, ECX
	DEC ECX
	CLD
	REP SCASB
	NOT ECX
	DEC ECX
	MOV EAX, ECX
	POP EDI
 %endmacro
 %define path DWORD [0x4B5B4C]
 strlen path

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
４．ラベルジャンプ

<ラベル名>:
という構文でラベルの設置が可能です。

 例

 XOR ECX, ECX
 LOOP_START:
	INC ECX
 	CMP ECX, 0x10
	JL LOOP_START
