### RETRET式コード生成の例 ###
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

RETRET式コードでは、paramsの値を直接コードに埋め込むことにより、
mugenの記述からコードへの値の受け渡しが可能です。
下記の例では、実際にparamsを用いてコードへ値を受け渡しています。

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
・使用するアセンブリコードの例
（paramsにより指定したスロットのPアドレスをsysvar(0)に保存するコードの例）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

PUSHAD
MOV EDI, DWORD [0x4B5B4C]        ;コード中に"%c"を利用できるため、数個までなら0x00などの使用不可文字の使用も許容できる
XOR ECX, ECX
MOV CL, 0x21                     ;0x21はparams利用のためのプレースホルダ（※）、後でここにPアドレスを取得する対象スロット番号（0,1,2,...）を埋め込む
XOR EDX, EDX
MOV DX, 0xB754
LEA EDX, [EDX+ECX*0x4]
MOV EDI, DWORD [EDI+EDX]
MOV DX, 0xFD0
MOV DWORD [ESI+EDX], EDI
POPAD
ADD ESP, 0xC
POP EDI
XOR EAX,EAX
POP ESI
RET

※プレースホルダとして使用する文字は、%cを利用して置き換えられる文字であれば何でもよい

ここから、出力結果のparamsを変更しコードへ値を埋め込んでいきます。

上記のコードでは、
paramsの1個目が、"MOV EDI,DWORD PTR [0x4B5B4C]"中に含まれる0x00に、
paramsの2個目が、"MOV CL,0x21"中に含まれる0x21(=33)に、
paramsの3個目、4個目が復帰処理などに使用される予約paramsとなっています。

そこでparamsを

params=0,ifelse(!enemy,alive&&numenemy>1,4-teamside,2-teamside),0,0

などと書き換えることで、
「普段はenemy(0)のPアドレスを取得するが、相手側1Pが死んでいて、
  かつ、相手側2Pが存在する場合のみenemy(1)のPアドレスを取得するコード」
のような複雑な挙動を取るコードも楽に作成できます。
